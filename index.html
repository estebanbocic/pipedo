Optional Supabase setup

The app is fully functional without Supabase. To sync, run this SQL in a Supabase project, then place the project URL and anon key in config.js.

DEMO SECURITY WARNING: These permissive RLS policies let anyone with the public anon key read, create, edit, and delete every task. Use them only for a personal demo or trusted environment. Add authentication and user-scoped policies before exposing real or sensitive data.

create table if not exists public.tasks (
  id text primary key,
  title text not null,
  description text not null default '',
  status text not null check (status in ('backlog', 'todo', 'in_progress', 'done')),
  color text not null default 'neutral',
  due_date date,
  position bigint not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint pipedo_tasks_color_check check (color in ('neutral', 'blue', 'cyan', 'violet', 'pink', 'green'))
);

alter table public.tasks add column if not exists color text;
alter table public.tasks add column if not exists due_date date;
update public.tasks
  set color = 'neutral'
  where color is null or color not in ('neutral', 'blue', 'cyan', 'violet', 'pink', 'green');
alter table public.tasks alter column color set default 'neutral';
alter table public.tasks alter column color set not null;

do $$
begin
  if not exists (
    select 1
    from pg_constraint
    where conrelid = 'public.tasks'::regclass
      and conname = 'pipedo_tasks_color_check'
  ) then
    alter table public.tasks
      add constraint pipedo_tasks_color_check
      check (color in ('neutral', 'blue', 'cyan', 'violet', 'pink', 'green'));
  end if;
end
$$;

alter table public.tasks enable row level security;

grant usage on schema public to anon;
grant select, insert, update, delete on public.tasks to anon;

drop policy if exists "pipedo_demo_select" on public.tasks;
drop policy if exists "pipedo_demo_insert" on public.tasks;
drop policy if exists "pipedo_demo_update" on public.tasks;
drop policy if exists "pipedo_demo_delete" on public.tasks;

create policy "pipedo_demo_select" on public.tasks
  for select to anon using (true);
create policy "pipedo_demo_insert" on public.tasks
  for insert to anon with check (true);
create policy "pipedo_demo_update" on public.tasks
  for update to anon using (true) with check (true);
create policy "pipedo_demo_delete" on public.tasks
  for delete to anon using (true);
